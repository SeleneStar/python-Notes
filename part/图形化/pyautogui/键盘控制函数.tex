\chapter{键盘控制函数}

\section{键盘输入}
\subsection{write() 输入文本}
\codeinline{write()} 函数模拟键盘输入文本。
\begin{lstlisting}[style=demostyle,numbers=none]
# text 是要输入的文本
# interval 是每个字符之间的间隔时间（秒）, 默认 0.0(s)
pyautogui.write(text)
>> pyautogui.write('Hello world!')                  # 输入 "Hello world!"，没有间隔时间
>>> pyautogui.write('Hello world!', interval=0.25)  # 输入 "Hello world!"，每个字符之间间隔 0.25 秒
\end{lstlisting}

\section{键盘按键}
\subsection{单键按键}
\begin{enumerate}[label=\arabic*)]
    \item \codeinline{keyDown()} 按下键盘按键
    \item \codeinline{keyUp()} 松开键盘按键
    \item \codeinline{press()} 按下并松开键盘按键
\end{enumerate}
\codeinline{press()} 函数是 \codeinline{keyDown()} 和 \codeinline{keyUp()} 的组合。
\begin{lstlisting}[style=demostyle,numbers=none]
# key 是要按下的键
# interval 是按下和松开之间的间隔时间（秒）, 默认 0.0(s)
# presses 是按键的次数, 默认 1
pyautogui.keyDown(key)
pyautogui.keyUp(key)
pyautogui.press(key, interval=0.0)  

>>> pyautogui.keyDown('shift')  # 持续按 左shift 键
>>> pyautogui.press('left')     # 按左箭头键
>>> pyautogui.press('left')     # 按左箭头键
>>> pyautogui.press('left')     # 按左箭头键
>>> pyautogui.keyUp('shift')    # 松开 左shift 键
>>> pyautogui.press('left', presses=3)  # 连续按三次左箭头键

# 如果连续按键可以使用列表
>>> pyautogui.press(['left', 'left', 'left'])   # 连续按三次左箭头键
\end{lstlisting}

\subsection{hold() 按住键盘按键}
\codeinline{hold()} 函数按住键盘按键，直到调用 \codeinline{release()} 函数松开。
\begin{lstlisting}[style=demostyle,numbers=none]
# key 是要按住的键
pyautogui.hold(key)

>>> with pyautogui.hold('shift'):
        pyautogui.press(['left', 'left', 'left'])

# 等同于
>>> pyautogui.keyDown('shift')  # hold down the shift key
>>> pyautogui.press('left')     # press the left arrow key
>>> pyautogui.press('left')     # press the left arrow key
>>> pyautogui.press('left')     # press the left arrow key
>>> pyautogui.keyUp('shift')    # release the shift key
\end{lstlisting}

\subsection{hotkey() 组合按键}
\codeinline{hotkey()} 函数模拟组合按键操作。
\begin{lstlisting}[style=demostyle,numbers=none]
# keys 是要组合按下的键，可以是单个键或多个键
# interval 是按下和松开之间的间隔时间（秒）, 默认 0
pyautogui.hotkey('ctrl', 'c')   # 按下 Ctrl + C 组合键


>>> pyautogui.hotkey('ctrl', 'shift', 'esc')
# 等同于
>>> pyautogui.keyDown('ctrl')
>>> pyautogui.keyDown('shift')
>>> pyautogui.keyDown('esc')
>>> pyautogui.keyUp('esc')
>>> pyautogui.keyUp('shift')
>>> pyautogui.keyUp('ctrl')
\end{lstlisting}
